import java.io.*;
import java.util.*;
import java.text.*;
import java.math.*;


public class Solution {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        if (n == 0) {
            System.out.println("No Second Highest Book Found");
            System.out.println("No Books Within Budget");
            return;
        }
        Book [] arr =  new Book[n];
        for (int i = 0; i < n; i++) {
            int a = sc.nextInt();
            sc.nextLine();
            String b =  sc.nextLine();
            String c = sc.nextLine();
            double d = sc.nextDouble();

            Book bk =  new Book(a, b, c, d);
            arr[i] =  bk;
        }
        double bud = sc.nextDouble();

        //find the book with highest price
        Book bkSecondHighestPrice = findSecondHighestPriceBook(arr);
        if(bkSecondHighestPrice != null) {
            System.out.println(bkSecondHighestPrice.getBookId() + " " + bkSecondHighestPrice.getTitle() + " "+ bkSecondHighestPrice.getPrice());
        } else {
            System.out.println("No Second Highest Book Found");
            
        }
        List<Book> answer = getBooksWithinBudget(arr, bud);
        if(answer==null) {
            System.out.println("No Books Within Budget");
        } else {
            for (Book bk : answer) {
                System.out.println(bk.getBookId() + " " + bk.getTitle() + " " + bk.getAuthor() + " " + bk.getPrice());
                
            }
        }
    }

    // static 
    static Book findSecondHighestPriceBook (Book[] arr) {
        if (arr == null || arr.length <= 1) {return null;}
        // find the book with second highest price
        int highestPriceIndex = 0;
        int secondHighestPriceIndex = -1;
        for (int i = 1; i < arr.length; i++) {

            //i is greater than hp
            if(arr[i].getPrice() > arr[highestPriceIndex].getPrice()) {
                //check for sp : if sp se bada hai toh hi matter karta 
               
                    secondHighestPriceIndex = highestPriceIndex;
                    highestPriceIndex =  i;
                
            }


            //i is less than hp
            if(arr[i].getPrice() < arr[highestPriceIndex].getPrice()) {
                //check for sp
                if(secondHighestPriceIndex == -1 || arr[i].getPrice() > arr[secondHighestPriceIndex].getPrice()) {
                    secondHighestPriceIndex = i;
                }
            }
        }

        if(secondHighestPriceIndex != -1) {
            return arr[secondHighestPriceIndex];
        } else {
            return null;
        }
    }
    static List<Book> getBooksWithinBudget (Book[] arr, double a) {
        List<Book> result =  new ArrayList<>();
        for (Book b : arr) {
            if (b.getPrice() <= a) {
                result.add(b);
            }
        }

        // sort the result in ascending order
        Collections.sort(result, (x, y) -> {
            int c = Double.compare(x.getPrice(), y.getPrice());
            if (c != 0) return c;
            return x.getTitle().compareToIgnoreCase(y.getTitle());
        });
        if (result.isEmpty()) {
            return null;
        } else {
            return result;
        }
    }
}

class Book {

    private int bookId;
    private String title;
    private String author;
    private double price;
    
    Book(int a, String b, String c, double d) {
        this.bookId = a;
        this.title = b;
        this.author = c;
        this.price = d;
    }

    //getters
    public int getBookId() {return this.bookId;}
    public String getTitle() {return this.title;}
    public String getAuthor() {return this.author;}
    public double getPrice() {return this.price;}


    //setters
    public void setBookId(int a) {this.bookId = a;}
    public void setTitle(String b) {this.title = b;}
    public void setAuthor(String c) {this.author = c;}
    public void setPrice(double d) {this.price = d;}
}
