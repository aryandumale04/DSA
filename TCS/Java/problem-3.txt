import java.io.*;
import java.util.*;
import java.util.regex.*;
import java.text.*;
import java.math.*;


public class Solution {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n =  sc.nextInt();
        if (n == 0) {
            System.out.println("No Hotel With Wifi Found");
            System.out.println("No Hotel Found");
            return;
        }
        Hotel[] arr =  new Hotel[n];
        for (int i = 0; i < n; i++) {
           int a =  sc.nextInt();
           sc.nextLine();
           String b = sc.nextLine();
           String c = sc.nextLine();
           boolean d =  sc.nextBoolean();
            sc.nextLine();
           double e =  sc.nextDouble();

          Hotel hos = new Hotel(a, b, c, d, e);
          arr[i] = hos;
        }
        sc.nextLine();
        String ct = sc.nextLine();
        int count = countHotelsWithWifiInCity(arr, ct);
        if (count == 0) {
            System.out.println("No Hotel With Wifi Found");
        } else {
            System.out.println(count);
        }
        double avgRent =  getAverageRentByCity(arr, ct);
        if (avgRent == 0.0) {
            System.out.println("No Hotel Found");
        } else {
            System.out.println(avgRent);
        }
        
    }

    //static methods
    static int countHotelsWithWifiInCity (Hotel[] arr, String ct) {
        if (arr == null) {return 0;}
        int count = 0;
        for (Hotel h : arr) {
            if (h.getCity().equalsIgnoreCase(ct) && h.getWifiAvailable() == true) {
                count++;
            }
        }

        return count;
    }
    static double getAverageRentByCity(Hotel[] arr, String ct) {
        if (arr == null) {
            return 0.0;
        }
        double avg = 0.0;
        double rent = 0;
        int count = 0;
        for (Hotel hk : arr) {
            if(hk.getCity().equalsIgnoreCase(ct)) {
                rent += hk.getRoomRent();
                count ++;
            }
        }
        if (count == 0) {
            return 0.0;
        }
        avg = rent / count;
        return avg;
        
    }
}


class Hotel {

    private int hotelId;
    private String hotelName;
    private String city;
    private boolean wifiAvailable;
    private double roomRent;
    
    Hotel(int a, String b, String c, boolean d, double e) {
        this.hotelId = a;
        this.hotelName = b;
        this.city = c;
        this.wifiAvailable = d;
        this.roomRent = e;
    }

    //getters 
    public int getHotelId() {return this.hotelId;}
    public String getHotelName() {return this.hotelName;}
    public String getCity() {return this.city;}
    public boolean getWifiAvailable() {return this.wifiAvailable;}
    public double getRoomRent() {return this.roomRent;}


    //setters
    public void setHotelId( int a) {this.hotelId = a;}
    public void setHotelName(String b) {this.hotelName = b;}
    public void setCity(String c) {this.city = c;}
    public void setWifiAvailable(boolean d) {this.wifiAvailable = d;}
    public void setRoomRent(double e) {this.roomRent = e;}
}
