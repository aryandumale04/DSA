import java.io.*;
import java.util.*;
import java.text.*;
import java.math.*;
import java.util.regex.*;

public class Solution {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();

        if (n == 0) {
            System.out.println("No Employee Found");
            return;
        }
        Employee[] emp =  new Employee[n];
        for(int i = 0 ; i < n; i++) {
            int id;
            String na;
            String d;
            double s;
            id =  sc.nextInt();
            sc.nextLine();
            na = sc.nextLine();
            d = sc.nextLine();
            
            s =  sc.nextDouble();
            emp[i] =  new Employee(id, na, d, s);   
        }
        sc.nextLine();
        String dep =  sc.nextLine();
        double dd = sc.nextDouble();

        Employee[] result = updateSalaryByDepartment(emp, dep, dd);
        if(result != null) {
            for (Employee e : result) {
                System.out.print(e.getEmpId() + " " + e.getEmpName() + " " + e.getDepartment() + " " + e.getSalary());
                System.out.println();

            }
        } else {
            System.out.println("No Employee Found");
        }
        
    }

    public static Employee[] updateSalaryByDepartment(Employee[] arr, String d, double per) {
        if (arr == null || arr.length == 0) return null;
        List<Employee> arr2 =  new ArrayList<>();
        for (Employee e : arr) {
            if (e.getDepartment().equalsIgnoreCase(d)) {
                //update the salary of this employee
                e.setSalary(e.getSalary() + ( (e.getSalary()/100) * per) );
                arr2.add(e);
            }
        }
        if (arr2.isEmpty()) {
            return null;
        } else {
            return arr2.toArray(new  Employee[0]);
        }
        
    }
}
class Employee {
    private int empId;
    private String empName;
    private String department;
    private double salary;

    Employee(int e, String n, String d, double s) {
        this.empId = e;
        this.empName = n;
        this.department = d;
        this.salary = s;
    }

    //getters
    public int getEmpId() {return this.empId;}
    public String getEmpName() {return this.empName;}
    public String getDepartment() {return this.department;}
    public double getSalary() {return this.salary;}

    //setters
    public void setEmpId(int e) {this.empId = e;}
    public void setEmpName(String s) {this.empName = s;}
    public void setDepartment(String d) {this.department = d;}
    public void setSalary(double s) {this.salary = s;}
}
